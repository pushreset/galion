# Coolify "Docker Compose" resource: TURN relay (coturn) + credential, quick-match and leaderboard service (ice) for galion
# Env vars to set in Coolify:
#   PUBLIC_IP   = public IP of the server
#   TURN_HOST   = domain or public IP clients use to reach coturn
#   TURN_SECRET = long random string (openssl rand -hex 32)
# Firewall: open 3478 UDP+TCP and 49160-49200 UDP directly on the server (not through Traefik).
# The world leaderboard lives in the `scores` volume (/data/scores.json): it survives redeploys.
volumes: # up here, not below: qa/quick.mjs runs everything after `content: |` as the service's script
  scores:
services:
  coturn:
    image: coturn/coturn:4.6
    network_mode: host
    restart: unless-stopped
    command: >
      -n --log-file=stdout
      --listening-port=3478 --min-port=49160 --max-port=49200
      --external-ip=${PUBLIC_IP} --realm=galion
      --use-auth-secret --static-auth-secret=${TURN_SECRET}
      --fingerprint --no-cli --no-tls --no-dtls
      --total-quota=20 --user-quota=8 --max-bps=1000000
      --denied-peer-ip=10.0.0.0-10.255.255.255
      --denied-peer-ip=172.16.0.0-172.31.255.255
      --denied-peer-ip=192.168.0.0-192.168.255.255
      --denied-peer-ip=127.0.0.0-127.255.255.255

  ice:
    image: node:22-alpine
    restart: unless-stopped
    command: node /app/ice.mjs
    environment:
      - SERVICE_FQDN_ICE_3000          # Coolify: domain + HTTPS via Traefik
      - TURN_SECRET=${TURN_SECRET}
      - TURN_HOST=${TURN_HOST}
      - ORIGIN=https://pushreset.github.io
    volumes:
      - scores:/data
      - type: bind
        source: ./ice.mjs
        target: /app/ice.mjs
        content: |
          // 1-hour coturn credentials (TURN REST API), quick-match queues and the world leaderboard, for the game's origin only
          import { createServer } from 'node:http';
          import { createHmac } from 'node:crypto';
          import { readFileSync, writeFileSync, renameSync } from 'node:fs';
          const { TURN_SECRET, TURN_HOST, ORIGIN, SCORES = '/data/scores.json' } = process.env;
          const hits = new Map(), posts = new Map(); // ponytail: in-memory per-IP limits, reset every 10 min
          setInterval(() => { hits.clear(); posts.clear(); }, 600_000);
          // World leaderboard: top 100 per game, ranked like the game's own board (wave, then gold).
          // ponytail: a score is whatever the client says (the host's browser runs the game, the server never sees it), so bounds and a
          // per-IP limit are the only guards: a server-checked replay if cheating shows up. One JSON file rewritten on each score: one instance, 200 rows
          const boards = { waves: [], convoy: [] };
          let onDisk = ''; // the JSON last written: a signing that changes nothing (a run off the board, a second signing) is not written again
          try { const saved = JSON.parse(readFileSync(SCORES, 'utf8')); for (const b in boards) if (Array.isArray(saved[b])) boards[b] = saved[b]; } catch {} // first start, or a broken file: empty boards
          const int = (v, lo, hi) => Number.isInteger(v) && v >= lo && v <= hi;
          // a run { ini, wave, gold, combo, ship } → { top, rank } (rank: its row, -1 off the board), null for a bad one
          function sign(b, s) {
            if (!(typeof s?.ini === 'string' && /^[A-Z]{1,3}$/.test(s.ini) && typeof s.ship === 'string' && /^[A-Za-z]{1,12}(\+[A-Za-z]{1,12}){0,3}$/.test(s.ship)
              && int(s.wave, 1, 999) && int(s.gold, 0, 1e6) && int(s.combo, 0, 999))) return null;
            const now = Date.now(), list = boards[b];
            // co-op: each browser of the crew signs the same run, they share one row; a solo run always gets its own
            // ponytail: a crew is told by its ships and score (two crews with the same ones within 10 min would share a row): the room code if that ever shows up
            let row = s.ship.includes('+') && list.find((r) => r.wave === s.wave && r.gold === s.gold && r.ship === s.ship && now - r.date < 600_000);
            if (row) {
              const inis = row.ini.split('+');
              if (!inis.includes(s.ini) && inis.length < 4) row.ini += '+' + s.ini;
              row.combo = Math.max(row.combo, s.combo);
            } else {
              row = { ini: s.ini, wave: s.wave, gold: s.gold, combo: s.combo, ship: s.ship, date: now };
              list.push(row);
              list.sort((x, y) => y.wave - x.wave || y.gold - x.gold); // stable: on a tie the earlier run stays ahead
              list.length = Math.min(list.length, 100);
            }
            const json = JSON.stringify(boards);
            if (json !== onDisk) {
              try { writeFileSync(SCORES + '.tmp', json); renameSync(SCORES + '.tmp', SCORES); onDisk = json; } // rename: a crash mid-write leaves the old file
              catch (e) { console.error('scores not saved', e.message); } // no volume, disk full: the board goes on in memory
            }
            return { top: list, rank: list.indexOf(row) };
          }
          // Quick match: one open room per queue; the server only hands out room codes, games stay P2P.
          // ponytail: rooms fill one at a time, in memory (a restart empties the queues, running games go on); region / skill buckets if traffic grows
          const SEATS = 3; // guests per room: MAX_PLAYERS - 1 in net.js
          const rooms = { pvp: null, 'coop-waves': null, 'coop-convoy': null }; // { code, free, seen, ready, handed }
          const newCode = () => Array.from({ length: 4 }, () => 'ABCDEFGHJKLMNPQRSTUVWXYZ'[Math.floor(Math.random() * 24)]).join('');
          // ?q=Q[&skip=CODE] → { host } | { join } | { retry }; ?q=Q&code=CODE&free=N → host heartbeat (N = 0 closes), null
          function match(q, p) {
            const now = Date.now(), code = p.get('code');
            let r = rooms[q];
            if (r && now - r.seen > (r.ready ? 25_000 : 10_000)) r = rooms[q] = null; // a host gone quiet, or one that never opened its room: a waiting host's heartbeat can take the slot
            if (code) {
              const free = Math.max(0, Math.min(SEATS, parseInt(p.get('free'), 10) || 0));
              if (r?.code === code) { if (free) Object.assign(r, { free: now - r.handed < 8000 ? Math.min(r.free, free) : free, seen: now, ready: true }); else rooms[q] = null; } // a seat just handed out: its guest is still connecting, the host does not count it yet
              else if (!r && free && /^[A-Z]{4}$/.test(code)) rooms[q] = { code, free, seen: now, ready: true }; // dropped (skip, full, timeout, restart) but still waiting: relist
              return null;
            }
            if (r?.code === p.get('skip')) rooms[q] = null; // a guest reports a dead / full room
            const o = rooms[q];
            if (o && !o.ready) return { retry: 1 }; // its host is still registering on PeerJS: a guest now would not find it
            if (o?.free > 0) { o.free--; o.handed = now; return { join: o.code }; }
            // every seat handed out (guests still connecting): open the next room
            rooms[q] = { code: newCode(), free: SEATS, seen: now, ready: false };
            return { host: rooms[q].code };
          }
          createServer((req, res) => {
            if (req.headers.origin !== ORIGIN) { res.writeHead(403).end(); return; }
            const cors = { 'Access-Control-Allow-Origin': ORIGIN, 'Cache-Control': 'no-store' };
            if (!URL.canParse(req.url, 'http://x')) { res.writeHead(400, cors).end(); return; } // '//' and the like: new URL would throw and take the process down
            const url = new URL(req.url, 'http://x'), q = url.searchParams.get('q');
            const ip = String(req.headers['x-forwarded-for'] || req.socket.remoteAddress).split(',').at(-1).trim(); // the last entry is the one Traefik wrote: those before it come from the client, who could pick a new address for each request
            if (!(url.pathname === '/match' && url.searchParams.get('code'))) { // heartbeats (every 10 s) don't count; an empty code is a lookup
              hits.set(ip, (hits.get(ip) || 0) + 1);
              if (hits.get(ip) > 60) { res.writeHead(429, cors).end(); return; }
            }
            if (url.pathname === '/scores') { // GET ?b=waves|convoy → the board; POST, the run as JSON → signs it
              const b = url.searchParams.get('b');
              if (!Object.hasOwn(boards, b)) { res.writeHead(400, cors).end(); return; }
              const send = (o) => { res.writeHead(o ? 200 : 400, { ...cors, 'Content-Type': 'application/json' }); res.end(o ? JSON.stringify(o) : ''); };
              if (req.method !== 'POST') { send({ top: boards[b], rank: -1 }); return; }
              posts.set(ip, (posts.get(ip) || 0) + 1);
              if (posts.get(ip) > 12) { res.writeHead(429, cors).end(); return; } // a run takes minutes: 12 covers four players behind one address
              let body = '';
              req.on('data', (d) => { body += d; if (body.length > 1024) req.destroy(); }); // a run is under 100 bytes
              req.on('end', () => { let s = null; try { s = JSON.parse(body); } catch {} send(sign(b, s)); });
              return;
            }
            if (url.pathname === '/match') {
              if (!Object.hasOwn(rooms, q)) { res.writeHead(400, cors).end(); return; }
              const m = match(q, url.searchParams);
              if (m?.retry) hits.set(ip, hits.get(ip) - 1); // changed nothing: 1 Hz polls for an opening room don't eat the budget shared with the credentials
              if (!m) { res.writeHead(204, cors).end(); return; }
              res.writeHead(200, { ...cors, 'Content-Type': 'application/json' });
              res.end(JSON.stringify(m));
              return;
            }
            const username = `${Math.floor(Date.now() / 1000) + 3600}:galion`;
            const credential = createHmac('sha1', TURN_SECRET).update(username).digest('base64');
            res.writeHead(200, { ...cors, 'Content-Type': 'application/json' });
            res.end(JSON.stringify({ iceServers: [
              { urls: 'stun:stun.l.google.com:19302' },
              { urls: [`turn:${TURN_HOST}:3478?transport=udp`, `turn:${TURN_HOST}:3478?transport=tcp`], username, credential },
            ] }));
          }).listen(3000);
